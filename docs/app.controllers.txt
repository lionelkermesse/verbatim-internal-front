@RestController
@RequestMapping("/api/event")
@RequiredArgsConstructor
@Tag(name = "Event Management", description = "Endpoints for managing session events.")
public class EventController {

    @PostMapping("/hierarchy")
    @Operation(summary = "")
    public ResponseEntity<ResultDto<ParsedEventDto>> hierarchy(
        @Parameter(description = "The event EVT file to upload for parsing")
        @RequestParam("file") MultipartFile file) throws IOException {

        final var result = EventParserHelper.parse(file.getInputStream());
        return ResponseEntity.ok(ResultDto.<ParsedEventDto>builder()
            .success(true)
            .data(result)
            .message("Event hierarchy parsed successfully.")
            .build());
    }

    /**
     * Parses an uploaded verbatim DOCX file and returns its plain text content directly.
     * This endpoint is for preview purposes and does not save the file or its content.
     *
     * @param file The verbatim DOCX file to parse.
     * @return A string with the verbatim's parsed text content.
     */
    @PostMapping("/parse")
    @Operation(summary = "Parses an event EVT and returns the content in JSON")
    public ResponseEntity<ResultDto<ParsedEventDto>> parse(
        @Parameter(description = "The event EVT file to upload for parsing")
        @RequestParam("file") MultipartFile file) throws IOException {

        final var result = EventParserHelper.parse(file.getInputStream(), true);
        return ResponseEntity.ok(ResultDto.<ParsedEventDto>builder()
            .success(true)
            .data(result)
            .message("Event parsed successfully.")
            .build());
    }

}


@Slf4j
@RestController
@RequestMapping("/api/v1/matching")
@RequiredArgsConstructor
public class MatchingController {
    private final SessionService sessionService;
    private final MatchingVersionService matchingVersionService;

    @PostMapping("/{sessionIdentifier}/start")
    @Operation(summary = "Upload a verbatim Word document for a specific session")
    public ResponseEntity<ResultDto<MatchingVersionDto>> start(
        @PathVariable String sessionIdentifier,
        @RequestParam(value = "verbatim") MultipartFile verbatimFile) throws IOException {
        log.info("Uploading verbatim Word document for a specific session: {}", sessionIdentifier);

        ValidationResult validation = SessionControllerValidator.validateVerbatimFile(verbatimFile);
        if (!validation.isValid()) {
            return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                .body(ResultDto.<MatchingVersionDto>builder()
                    .success(false)
                    .message(validation.getMessage())
                    .build());
        }

        final var result = matchingVersionService.startMatching(sessionIdentifier, verbatimFile);
        return ResponseEntity.ok(ResultDto.<MatchingVersionDto>builder()
            .success(true)
            .data(result)
            .message("Verbatim document uploaded successfully.")
            .build());
    }

    @PutMapping("/{sessionIdentifier}/update")
    @Operation(summary = "Update a single intervention's match manually")
    public ResponseEntity<ResultDto<MatchingVersionDto>> update(
        @PathVariable String sessionIdentifier, @Valid @RequestBody UpdateMatchRequest request) throws UpdateMatchException {
        log.info("Updating intervention match for session: {}", sessionIdentifier);

        final var updated = matchingVersionService.updateVersion(sessionIdentifier, request);
        return ResponseEntity.ok(ResultDto.<MatchingVersionDto>builder()
            .success(true)
            .data(updated)
            .message("Intervention match updated successfully.")
            .build());
    }

    @PostMapping("/{sessionIdentifier}/{version}/validate")
    @Operation(summary = "Validate and finalize the matching for a session")
    public ResponseEntity<ResultDto<MatchingVersionDto>> validate(@PathVariable String sessionIdentifier, @PathVariable Integer version) {
        final var result = matchingVersionService.validateVersion(sessionIdentifier, version);
        return ResponseEntity.ok(ResultDto.<MatchingVersionDto>builder()
            .success(true)
            .data(result)
            .message("Matching version validated successfully.")
            .build());
    }

    @GetMapping("/{sessionIdentifier}")
    @Operation(summary = "Get matching results for correction")
    public ResponseEntity<ResultDto<List<MatchingVersionDto>>> getAll(@PathVariable String sessionIdentifier) {
        log.info("Fetching matching results for session: {}", sessionIdentifier);

        List<MatchingVersionDto> results = matchingVersionService.getAllVersions(sessionIdentifier);
        return ResponseEntity.ok(ResultDto.<List<MatchingVersionDto>>builder()
            .success(true)
            .data(results)
            .message("Matching results fetched successfully.")
            .build());
    }

    @DeleteMapping("/{sessionIdentifier}/{version}")
    @Operation(summary = "Cancel/remove a specific matching version for a session")
    public ResponseEntity<ResultDto<Void>> remove(@PathVariable String sessionIdentifier, @PathVariable Integer version) {
        log.info("Removing matching version {} for session {}", version, sessionIdentifier);

        matchingVersionService.deleteMatchingVersion(sessionIdentifier, version);
        return ResponseEntity.ok(ResultDto.<Void>builder()
            .success(true)
            .message("Matching version removed successfully.")
            .build());
    }

    @DeleteMapping("/{sessionIdentifier}/version/latest")
    @Operation(summary = "Cancel/remove the latest matching version for a session")
    public ResponseEntity<ResultDto<Void>> removeLatest(@PathVariable String sessionIdentifier) {
        log.info("Removing latest matching version for session {}", sessionIdentifier);

        matchingVersionService.deleteLatestMatchingVersion(sessionIdentifier);
        return ResponseEntity.ok(ResultDto.<Void>builder()
            .success(true)
            .message("Latest matching version removed successfully.")
            .build());
    }

    @GetMapping("/{sessionIdentifier}/{version}/export")
    @Operation(summary = "Export a matching result for a session and version in JSON, XML, or CSV format")
    public ResponseEntity<Resource> export(
        @PathVariable String sessionIdentifier,
        @PathVariable Integer version,
        @RequestParam FileContentType format) throws IOException {
        log.info("Exporting matching result for session {} and version {} to format '{}'", sessionIdentifier, version, format);

        final var filename = String.format("matching_%s_v%d.%s", sessionIdentifier, version, format.name().toLowerCase());
        final var file = "attachment; filename=" + filename;
        final var exportedMatchingResult = matchingVersionService.exportMatchingResult(sessionIdentifier, version, format);
        final var content = exportedMatchingResult.getBytes(StandardCharsets.UTF_8);
        return ResponseEntity.ok()
            .header(HttpHeaders.CONTENT_DISPOSITION, file)
            .header(HttpHeaders.CONTENT_TYPE, switch (format) {
                case JSON -> MediaType.APPLICATION_JSON_VALUE;
                case XML -> MediaType.APPLICATION_XML_VALUE;
                case CSV -> "text/csv";
            })
            .body(new ByteArrayResource(content));
    }

@Slf4j
@RestController
@RequestMapping("/api/v1/referential/files")
@RequiredArgsConstructor
@Tag(name = "Referential Files", description = "APIs for managing referential files (XML/JSON)")
public class ReferentialFileController {

    private final ReferentialFileService referentialFileService;

    @PostMapping(consumes = MediaType.MULTIPART_FORM_DATA_VALUE)
    @Operation(summary = "Upload a new referential file (XML or JSON format)")
    public ResponseEntity<ResultDto<ReferentialFileDto>> uploadFile(
        @Parameter(description = "Referential file to upload (XML or JSON format)")
        @RequestParam("file") MultipartFile file) throws IOException {

        String username = SecurityUtils.getCurrentUserLogin().orElse("system");

        ReferentialFileDto fileDto = referentialFileService.uploadFile(file, username);

        return ResponseEntity.ok(ResultDto.<ReferentialFileDto>builder()
            .success(true)
            .data(fileDto)
            .message("File uploaded successfully")
            .build());
    }

    @GetMapping("/{id}/download")
    @Operation(summary = "Download a referential file by ID")
    public ResponseEntity<Resource> downloadFile(
        @Parameter(description = "ID of the file to download")
        @PathVariable Long id) {

        Resource resource = referentialFileService.downloadFile(id);
        ReferentialFileDto fileDto = referentialFileService.getFileMetadata(id);

        String contentType = fileDto.getContentType() != null ?
            fileDto.getContentType() : "application/octet-stream";

        return ResponseEntity.ok()
            .contentType(MediaType.parseMediaType(contentType))
            .header(HttpHeaders.CONTENT_DISPOSITION,
                "attachment; filename=\"" + fileDto.getOriginalFilename() + "\"")
            .body(resource);
    }

    @GetMapping("/{id}")
    @Operation(summary = "Get file metadata by ID")
    public ResponseEntity<ResultDto<ReferentialFileDto>> getFile(
        @Parameter(description = "ID of the file to retrieve")
        @PathVariable Long id) {

        ReferentialFileDto fileDto = referentialFileService.getFileMetadata(id);
        return ResponseEntity.ok(ResultDto.<ReferentialFileDto>builder()
            .success(true)
            .data(fileDto)
            .build());
    }

    @GetMapping("/{id}/content")
    @Operation(summary = "Get parsed file content by ID")
    public ResponseEntity<ResultDto<ReferentialContentDto>> getFileContent(
        @Parameter(description = "ID of the file")
        @PathVariable Long id) throws IOException {

        ReferentialContentDto contentDto = referentialFileService.getFileContent(id);
        return ResponseEntity.ok(ResultDto.<ReferentialContentDto>builder()
            .success(true)
            .data(contentDto)
            .build());
    }

    @GetMapping
    @Operation(summary = "Get all referential files (without content)")
    public ResponseEntity<ResultDto<List<ReferentialFileDto>>> getAllFiles() {
        List<ReferentialFileDto> files = referentialFileService.getAllFiles();
        return ResponseEntity.ok(ResultDto.<List<ReferentialFileDto>>builder()
            .success(true)
            .data(files)
            .build());
    }

    @DeleteMapping("/{id}")
    @Operation(summary = "Delete a referential file by ID (soft delete)")
    public ResponseEntity<ResultDto<Void>> deleteFile(
        @Parameter(description = "ID of the file to delete")
        @PathVariable Long id) {

        referentialFileService.deleteFile(id);
        return ResponseEntity.ok(ResultDto.<Void>builder()
            .success(true)
            .message("File deleted successfully")
            .build());
    }

    @GetMapping("/current")
    @Operation(summary = "Get the current referential file")
    public ResponseEntity<ResultDto<ReferentialFileDto>> getCurrentReferential() {
        ReferentialFileDto currentReferential = referentialFileService.getLatestVersion();
        return ResponseEntity.ok(ResultDto.<ReferentialFileDto>builder()
            .success(true)
            .data(currentReferential)
            .build());
    }
}



@Slf4j
@RestController
@RequestMapping("/api/v1/sessions")
@RequiredArgsConstructor
public class SessionController {
    private final SessionService sessionService;

    @GetMapping
    @Operation(summary = "List available sessions for processing")
    public ResponseEntity<ResultDto<List<SessionDto>>> getAllSessions() {
        final var sessions = sessionService.getAll();
        return ResponseEntity.ok(ResultDto.<List<SessionDto>>builder()
            .success(true)
            .data(sessions)
            .message("Available sessions fetched successfully.")
            .build());
    }

    @PostMapping("/create")
    @Operation(summary = "Create a new session")
    public ResponseEntity<ResultDto<SessionDto>> createSession(@Valid @RequestBody SessionCreateDto sessionCreateDto) {
        final var created = sessionService.createSession(sessionCreateDto);
        return ResponseEntity.status(HttpStatus.CREATED)
            .body(ResultDto.<SessionDto>builder()
                .success(true)
                .data(created)
                .message("Session created successfully.")
                .build());
    }
}

@RestController
@RequestMapping("/api/verbatim")
@RequiredArgsConstructor
@Tag(name = "Verbatim Management", description = "Endpoints for uploading and managing verbatim documents.")
public class VerbatimController {

    /**
     * Parses an uploaded verbatim DOCX file and returns its plain text content directly.
     * This endpoint is for preview purposes and does not save the file or its content.
     *
     * @param file The verbatim DOCX file to parse.
     * @return A string with the verbatim's parsed text content.
     */
    @PostMapping("/parse")
    @Operation(summary = "Parses a verbatim DOCX and returns the content as JSON")
    public ResponseEntity<ParsedVerbatimDto> parse(
        @Parameter(description = "The verbatim DOCX file to upload for parsing")
        @RequestParam("file") MultipartFile file,
        @Parameter(description = "The session identifier of the verbatim file")
        @RequestParam("sessionIdentifier") String sessionIdentifier) throws IOException {

        final var parsedVerbatimDto = VerbatimParserHelper.parse(file.getInputStream(), sessionIdentifier);
        return ResponseEntity.ok().body(parsedVerbatimDto);
    }
}
