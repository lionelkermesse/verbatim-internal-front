@Getter
public enum MatchingStatus {
    /**
     * The match was made automatically by the algorithm.
     */
    AUTOMATICALLY_MATCHED(1L),
    /**
     * The match was made or corrected manually by a user.
     */
    MANUALLY_MATCHED(2L),
    /**
     * The intervention has no corresponding event.
     */
    UNMATCHED(3L),
    /**
     * The algorithm found multiple potential matches, requiring user intervention.
     */
    MATCHING_CONFLICT(4L);

    private final Long id;

    MatchingStatus(Long id) {
        this.id = id;
    }
}

    JSON,
    XML,
    CSV
}

@Getter
public enum VersionStatus {
    /**
     * Matching files have been uploaded, ready for processing.
     */
    UPLOADED(1L),
    /**
     * Matching process is running.
     */
    PROCESSING(2L),
    /**
     * Matching complete, waiting for user validation.
     */
    AWAITING_CORRECTION(3L),
    /**
     * User has validated the version.
     */
    VALIDATED(4L),
    /**
     * An error occurred during processing.
     */
    ERROR(5L);

    private static final Map<Long, VersionStatus> GET_BY_ID = Stream.of(values()).collect(Collectors.toMap(VersionStatus::getId, Function.identity()));

    private final Long id;

    VersionStatus(Long id) {
        this.id = id;
    }

    public static VersionStatus getById(Long id) {
        return GET_BY_ID.get(id);
    }
}

/**
 * Base abstract class for entities which will hold definitions for created, last modified, created by,
 * last modified by attributes.
 *
 * @author LKE
 * @created on 26.08.2025
 */
@Setter
@Getter
@MappedSuperclass
@EntityListeners(AuditingEntityListener.class)
@JsonIgnoreProperties(value = { "createdBy", "createdDate", "lastModifiedBy", "lastModifiedDate" }, allowGetters = true)
public abstract class AbstractAuditingEntity<T> implements Serializable {

    private static final long serialVersionUID = 1L;

    public abstract T getId();

    @CreatedBy
    @Column(name = "created_by", nullable = false, length = 50, updatable = false)
    protected String createdBy;

    @CreatedDate
    @Column(name = "created_date", updatable = false)
    protected Instant createdDate = Instant.now();

    @LastModifiedBy
    @Column(name = "last_modified_by", length = 50)
    protected String lastModifiedBy;

    @LastModifiedDate
    @Column(name = "last_modified_date")
    protected Instant lastModifiedDate;

    @PreUpdate
    public void preUpdate() {
        this.lastModifiedDate = Instant.now();
    }
}

    @JoinColumn(name = "session_id", nullable = false)

    private Integer version;

    /**
     * Stores a JSON snapshot of List<MatchingResultDto> of the interventions and their matched event IDs
     * at the moment of validation. Uses JSONB for PostgreSQL, CLOB for H2.
     */
    @Lob
    @Column(name = "matching_result_snapshot", columnDefinition = "clob")
    private String matchingResult;

    @Column(name = "created_at", nullable = false)
    private Instant createdAt;

    @Column(name = "created_by", nullable = false)
    private String createdBy;

    @Column(name = "validated_at")
    private LocalDateTime validatedAt;

    @Column(name = "validated_by")
    private String validatedBy;

    @Column(name = "status", nullable = false)
    @Convert(converter = VersionStatusConverter.class)
    private VersionStatus status = VersionStatus.UPLOADED;

    @PrePersist
    public void prePersist() {
        this.createdAt = Instant.now();
    }

    @PreUpdate
    public void preUpdate() {
        this.validatedAt = LocalDateTime.now();
    }
}

        return Integer.compare(this.rank, o.rank);

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class MatchingResultDto {
    @Builder.Default
    private List<MatchingResultItemDto> result = new ArrayList<>();
    @Builder.Default
    private Set<SpeakerDto> speakers = new HashSet<>();

    /**
     * Recursively collects all items with status AUTOMATICALLY_MATCHED or MANUALLY_MATCHED from the result tree.
     */
    @JsonIgnore
    public List<MatchingResultItemDto> getMatched() {
        List<MatchingResultItemDto> matchedItems = new ArrayList<>();
        collectByStatus(result, matchedItems, MatchingStatus.AUTOMATICALLY_MATCHED, MatchingStatus.MANUALLY_MATCHED);
        return matchedItems;
    }

    /**
     * Recursively collects all items with status UNMATCHED or MATCHING_CONFLICT from the result tree.
     */
    @JsonIgnore
    public List<MatchingResultItemDto> getUnMatched() {
        final List<MatchingResultItemDto> unmatchedItems = new ArrayList<>();
        collectByStatus(result, unmatchedItems, MatchingStatus.UNMATCHED, MatchingStatus.MATCHING_CONFLICT);
        return unmatchedItems;
    }

    public Set<SpeakerDto> getSpeakers() {
        if (speakers.isEmpty()) {
            this.speakers = result.stream()
                .map(MatchingResultItemDto::getSpeakers)
                .flatMap(Set::stream)
                .collect(Collectors.toSet());
        }

        return speakers;
    }

    /**
     * Helper method to recursively collect items by status.
     */
    private void collectByStatus(final List<MatchingResultItemDto> source, final List<MatchingResultItemDto> target, final MatchingStatus... statuses) {
        if (source == null) return;
        for (MatchingResultItemDto item : source) {
            for (MatchingStatus status : statuses) {
                if (item.getStatus() == status) {
                    target.add(item);
                    break;
                }
            }
            collectByStatus(item.getInners(), target, statuses);
        }
    }
}
@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class MatchingResultItemDto implements Comparable<MatchingResultItemDto> {
    private Long id;
    private String verbatim;
    private Integer lineNumber;
    private String title;
    @Builder.Default
    private List<MatchingResultItemDto> inners = new ArrayList<>();
    @Builder.Default
    private Set<SpeakerDto> speakers = new HashSet<>();
    @Builder.Default
    private MatchingStatus status = MatchingStatus.UNMATCHED;

    public Set<SpeakerDto> getSpeakers() {
        final Set<SpeakerDto> allSpeakers = new HashSet<>(speakers);
        inners.stream()
            .map(MatchingResultItemDto::getSpeakers)
            .flatMap(Collection::stream)
            .forEach(speaker -> SpeakerUtils.addSpeakerToList(speaker, allSpeakers));
        return allSpeakers;
    }

    public MatchingResultItemDto addSpeaker(final SpeakerDto newSpeaker) {
        if (newSpeaker != null && StringUtils.isNotBlank(newSpeaker.getSpeakerFullName())) {
            final var speakerMap = speakers.stream().collect(Collectors.toMap(SpeakerDto::getSpeakerFullName, Function.identity()));
            final var existingSpeaker = speakerMap.get(newSpeaker.getSpeakerFullName());
            final var mapper = Mappers.getMapper(SpeakerMapper.class);
            if (existingSpeaker != null) {
                mapper.partialUpdate(existingSpeaker, newSpeaker);
            } else {
                speakerMap.put(newSpeaker.getSpeakerFullName(), newSpeaker);
            }
            this.speakers = new HashSet<>(speakerMap.values());
        }
        return this;
    }

    @Override
    public int compareTo(@NotNull MatchingResultItemDto o) {
        // If ranks are equal, compare by ID with null safety
        if (this.id == null && o.getId() == null) {
            return 0;
        }
        if (this.id == null) {
            return -1; // null comes first
        }
        if (o.getId() == null) {
            return 1; // non-null comes after null
        }

        return this.id.compareTo(o.getId());
    }
}
@Data
public class MatchingRuleDto {
    @JsonProperty("verbatim_term")
    @CsvBindByName(column = "verbatim_term")
    private String verbatimTerm;

    @JsonProperty("event_term")
    @CsvBindByName(column = "event_term")
    private String eventTerm;

    @JsonProperty("type")
    @JsonSerialize(using = MatchingRuleTypeJsonConfig.Serializer.class)
    @JsonDeserialize(using = MatchingRuleTypeJsonConfig.Deserializer.class)
    @CsvCustomBindByName(column = "type", converter = MatchingRuleTypeConverter.class)
    private MatchingConfig.MatchingRuleType type;

    /**
     * Checks if this rule represents an exact match.
     * @return true if the rule type is EXACT
     */
    public boolean isExactMatch() {
        return type == MatchingConfig.MatchingRuleType.EXACT;
    }

    /**
     * Checks if this rule represents an approximate match.
     * @return true if the rule type is APPROXIMATE
     */
    public boolean isApproximateMatch() {
        return type == MatchingConfig.MatchingRuleType.APPROXIMATE;
    }

    /**
     * Checks if this rule represents a similarity match.
     * @return true if the rule type is SIMILARITY or null
     */
    public boolean isSimilarityMatch() {
        return type == null || type == MatchingConfig.MatchingRuleType.SIMILARITY;
    }
}

@Data
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class ParsedEventDto {
    /**
     * The date of the session, extracted from the document.
     */
    @JsonFormat(shape = JsonFormat.Shape.STRING, pattern = "dd-MM-yyyy")
    private LocalDate sessionDate;

    /**
     * The start time of the session, extracted from the document.
     */
    @JsonFormat(shape = JsonFormat.Shape.STRING, pattern = "HH:mm:ss")
    private LocalTime sessionStartTime;

    /**
     * The end time of the session, extracted from the document.
     */
    @JsonFormat(shape = JsonFormat.Shape.STRING, pattern = "HH:mm:ss")
    private LocalTime sessionEndTime;

    /**
     * The root events in the hierarchy, typically top-level verbatims.
     */
    @Builder.Default
    private List<EventDto> root = new ArrayList<>();

    /**
     * Events that could not be attached to the main hierarchy.
     */
    @Builder.Default
    private List<EventDto> orphans = new ArrayList<>();

    public ParsedEventDto addRootEvent(EventDto eventDto) {
        root.add(eventDto);
        return this;
    }

    public ParsedEventDto addOrphan(EventDto eventDto) {
        orphans.add(eventDto);
        return this;
    }

    @JsonIgnore
    public ParsedEventDto updateMedatadata() {
        // If there are root events, extract session metadata from the first and last root events.
        if (CollectionUtils.isNotEmpty(root)) {
            EventDto first = root.get(0);
            EventDto last = root.get(root.size() - 1);

            // Set session date if available
            if (first.getSessionDate() != null) {
                setSessionDate(first.getSessionDate().toLocalDate());
            }

            // Set session start time if available
            if (first.getSessionStart() != null) {
                setSessionStartTime(first.getSessionStart().toLocalTime());
            }

            // Set session end time if available
            if (last.getSessionEnd() != null) {
                setSessionEndTime(last.getSessionEnd().toLocalTime());
            }
        }

        return this;
    }
}

        final String speakerName = StringUtils.trimToEmpty(StringUtils.defaultIfBlank(firstName, "") + " " + StringUtils.defaultIfBlank(lastName, ""));

    }

        int lastNameComparison = String.valueOf(this.lastName).compareToIgnoreCase(String.valueOf(o.lastName));

    }
